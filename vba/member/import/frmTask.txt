Private Sub AddLbl(ByVal cap As String, ByVal y As Single)
    AddLblAt cap, 8, y

Private Sub AddLblAt(ByVal cap As String, ByVal x As Single, ByVal y As Single)
    With Me.Controls.Add("Forms.Label.1")
        .Caption = cap: .Left = x: .Top = y + 3: .Width = 66: .Height = 14
    End With
End Sub

Private Function AddTxt(ByVal nm As String, ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single) As MSForms.TextBox
    Set AddTxt = Me.Controls.Add("Forms.TextBox.1", nm)
    With AddTxt
        .Left = x: .Top = y: .Width = w: .Height = h
    End With
End Function

Private Function AddCbo(ByVal nm As String, ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal hook As Boolean) As MSForms.ComboBox
    Set AddCbo = Me.Controls.Add("Forms.ComboBox.1", nm)
    With AddCbo
        .Left = x: .Top = y: .Width = w: .Height = 18
        .Style = fmStyleDropDownList
    End With
    If hook Then HookCtl AddCbo, nm
End Function

Private Function AddBtn(ByVal nm As String, ByVal cap As String, ByVal x As Single, ByVal y As Single, ByVal w As Single) As MSForms.CommandButton
    Set AddBtn = Me.Controls.Add("Forms.CommandButton.1", nm)
    With AddBtn
        .Caption = cap: .Left = x: .Top = y: .Width = w: .Height = 22
    End With
    HookCtl AddBtn, nm
End Function

Private Sub HookCtl(ByVal ctl As Object, ByVal nm As String)
    Dim h As clsCtl
    Set h = New clsCtl
    Set h.Host = Me
    h.CtlName = nm
    Select Case TypeName(ctl)
        Case "CommandButton": Set h.Btn = ctl
        Case "ComboBox": Set h.Cbo = ctl
        Case "ListBox": Set h.Lst = ctl
    End Select
    mHooks.Add h
End Sub

'--- 값 채우기 ----------------------------------------------
Public Sub Init(ByVal id As String, Optional ByVal defDate As Date = 0, Optional ByVal defMile As Boolean = False)
    Dim v As Variant
    mLoading = True
    mID = id
    LoadProjects
    If id <> "" Then
        v = GetTaskArr(id)
        If IsEmpty(v) Then
            mID = ""
        Else
            Me.Caption = "Task 수정 - " & id
            SetCombo cboProject, CStr(Nz(v(T_PID)))
            If cboProject.ListIndex < 0 And CStr(Nz(v(T_PID))) <> "" Then   ' 종료된 프로젝트 등 목록에 없는 코드
                cboProject.AddItem CStr(v(T_PID))
                cboProject.List(cboProject.ListCount - 1, 1) = CStr(v(T_PID)) & "  " & CStr(Nz(v(T_PNAME)))
                SetCombo cboProject, CStr(v(T_PID))
            End If
            txtName.Text = CStr(Nz(v(T_NAME)))
            txtStart.Text = Format$(ToDate(v(T_START)), "yyyy-mm-dd")
            txtDue.Text = Format$(ToDate(v(T_DUE)), "yyyy-mm-dd")
            SetCombo cboPct, CStr(NumOr(v(T_PCT)))
            If cboPct.ListIndex < 0 Then cboPct.AddItem CStr(NumOr(v(T_PCT))): SetCombo cboPct, CStr(NumOr(v(T_PCT)))
            SetCombo cboStatus, CStr(Nz(v(T_STATUS), ST_PLAN))
            SetCombo cboPri, CStr(Nz(v(T_PRI), "중"))
            chkMile.Value = (UCase$(CStr(Nz(v(T_MILE)))) = "Y")
            txtNote.Text = CStr(Nz(v(T_NOTE)))
            If CStr(Nz(v(T_COMMENT))) <> "" Then
                lblComment.Caption = " " & SYM_COMMENT & " 팀장 코멘트: " & v(T_COMMENT)
                lblComment.Visible = True
            End If
            btnDelete.Visible = True
            mLastProject = CStr(Nz(v(T_PID)))
            LoadPhases mLastProject, CStr(Nz(v(T_PHASE)))
        End If
    End If
    If mID = "" Then
        If defDate = 0 Then defDate = Date
        txtStart.Text = Format$(defDate, "yyyy-mm-dd")
        txtDue.Text = Format$(defDate, "yyyy-mm-dd")
        SetCombo cboPct, "0"
        SetCombo cboStatus, ST_PLAN
        SetCombo cboPri, "중"
        chkMile.Value = defMile
        If defMile Then Me.Caption = "보고·마감 일정 등록"
        LoadPhases "", ""
    End If
    btnPhase.Visible = IsPM(mLastProject)
    mLoading = False
End Sub

Private Sub LoadProjects()
    Dim a As Variant, i As Long, n As Long, lst() As Variant, specials As Variant, k As Long
    a = ReadTable(GetLO(TB_PROJ))
    specials = Array(PJ_COMMON, PJ_LEAVE, PJ_TRIP, PJ_EDU)
    ReDim lst(0 To RowCount(a) + 5, 0 To 1)
    For i = 1 To RowCount(a)
        If IsActiveProject(CStr(Nz(a(i, P_STATUS)))) And CStr(Nz(a(i, P_ID))) <> "" Then
            lst(n, 0) = CStr(a(i, P_ID))
            lst(n, 1) = CStr(a(i, P_ID)) & IIf(CStr(a(i, P_STATUS)) = PJ_PENDING, "* ", "  ") & CStr(Nz(a(i, P_NAME)))
            n = n + 1
        End If
    Next i
    For k = 0 To 3
        lst(n, 0) = specials(k): lst(n, 1) = specials(k): n = n + 1
    Next k
    lst(n, 0) = "+": lst(n, 1) = "[+ 새 프로젝트 요청]": n = n + 1
    cboProject.Clear
    For i = 0 To n - 1
        cboProject.AddItem lst(i, 0)
        cboProject.List(i, 1) = lst(i, 1)
    Next i
End Sub

Private Sub LoadPhases(ByVal pid As String, ByVal selId As String)
    Dim ph As Variant, i As Long
    cboPhase.Clear
    cboPhase.AddItem "-"
    cboPhase.List(0, 1) = "(단계 미지정)"
    ph = GetPhases(pid)
    For i = 1 To RowCount(ph)
        cboPhase.AddItem CStr(ph(i, PH_ID))
        cboPhase.List(cboPhase.ListCount - 1, 1) = ph(i, PH_ORDER) & ". " & ph(i, PH_NAME)
    Next i
    If selId = "" Then selId = "-"
    SetCombo cboPhase, selId
    If cboPhase.ListIndex < 0 Then cboPhase.ListIndex = 0
    cboPhase.Enabled = (pid <> "" And Not IsSpecialProject(pid))
End Sub

Private Sub SetCombo(ByVal cbo As MSForms.ComboBox, ByVal key As String)
    Dim i As Long
    For i = 0 To cbo.ListCount - 1
        If CStr(cbo.List(i, 0)) = key Then cbo.ListIndex = i: Exit Sub
    Next i
End Sub

Private Function ComboKey(ByVal cbo As MSForms.ComboBox) As String
    If cbo.ListIndex >= 0 Then ComboKey = CStr(cbo.List(cbo.ListIndex, 0))
End Function

'--- 이벤트 (clsCtl에서 호출) -------------------------------
Public Sub OnCtlChange(ByVal nm As String)
    Dim pid As String, f As frmRequest, code As String
    If mLoading Then Exit Sub
    If nm <> "cboProject" Then Exit Sub
    pid = ComboKey(cboProject)
    If pid = "+" Then
        mLoading = True
        Set f = New frmRequest
        f.Show
        code = f.NewCode
        Unload f
        LoadProjects
        If code <> "" Then
            SetCombo cboProject, code
            pid = code
        Else
            SetCombo cboProject, mLastProject
            pid = mLastProject
        End If
        mLoading = False
    End If
    mLastProject = pid
    LoadPhases pid, ""
    btnPhase.Visible = IsPM(pid)
End Sub

Public Sub OnCtlClick(ByVal nm As String)
    Select Case nm
        Case "btnSave": DoSave
        Case "btnDelete": DoDelete
        Case "btnCancel": Me.Hide
        Case "btnPhase"
            If OpenPhaseForm(ComboKey(cboProject)) Then LoadPhases ComboKey(cboProject), ComboKey(cboPhase)
    End Select
End Sub

Private Sub DoSave()
    Dim v As Variant, s As Date, e As Date, pid As String, ph As String
    pid = ComboKey(cboProject)
    If pid = "" Or pid = "+" Then Msg "프로젝트를 선택하세요.", vbExclamation: Exit Sub
    If Trim$(txtName.Text) = "" Then Msg "업무명을 입력하세요.", vbExclamation: txtName.SetFocus: Exit Sub
    s = ParseDate(txtStart.Text)
    e = ParseDate(txtDue.Text)
    If s = 0 Then Msg "시작일 형식이 올바르지 않습니다. (예: 2026-10-07)", vbExclamation: txtStart.SetFocus: Exit Sub
    If e = 0 Then Msg "종료일 형식이 올바르지 않습니다. (예: 2026-10-14)", vbExclamation: txtDue.SetFocus: Exit Sub
    If e < s Then Msg "종료일이 시작일보다 빠릅니다.", vbExclamation: txtDue.SetFocus: Exit Sub

    If mID <> "" Then v = GetTaskArr(mID)
    If IsEmpty(v) Then v = NewTaskArr()
    ph = ComboKey(cboPhase)
    If ph = "-" Then ph = ""
    v(T_PID) = pid
    v(T_PHASE) = IIf(ph = "", Empty, ph)
    v(T_NAME) = Trim$(txtName.Text)
    v(T_START) = s
    v(T_DUE) = e
    v(T_PCT) = CDbl(Val(ComboKey(cboPct)))
    v(T_STATUS) = ComboKey(cboStatus)
    v(T_PRI) = ComboKey(cboPri)
    v(T_MILE) = IIf(chkMile.Value, "Y", Empty)
    v(T_NOTE) = IIf(Trim$(txtNote.Text) = "", Empty, Trim$(txtNote.Text))
    mSavedID = SaveTask(v)
    mSaved = True
    Me.Hide
End Sub

Private Sub DoDelete()
    If mID = "" Then Exit Sub
    If MsgBox("이 Task를 삭제할까요?" & vbLf & "(기록을 남기려면 삭제 대신 상태를 '취소'로 바꾸세요)", _
              vbYesNo + vbExclamation, APP_TITLE) <> vbYes Then Exit Sub
    DeleteTask mID
    mSavedID = ""
    mSaved = True
    Me.Hide
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If CloseMode = vbFormControlMenu Then
        Cancel = True
        Me.Hide
    End If
End Sub

Private Sub UserForm_Terminate()
    Set mHooks = Nothing
End Sub
